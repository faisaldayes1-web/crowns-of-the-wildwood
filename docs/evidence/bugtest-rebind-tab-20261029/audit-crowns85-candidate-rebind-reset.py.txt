read = lambda p: json.loads(Path(p).read_text(encoding='utf-8-sig'))
sha = lambda p: hashlib.sha256(Path(p).read_bytes()).hexdigest()

def tree_hashes(folder):
    p = Path(folder)
    return {str(f.relative_to(p)): sha(f) for f in p.rglob('*') if f.is_file()} if p.exists() else {}

def map_sets(value):
    return {k: sorted(json.dumps(x, sort_keys=True) for x in v) for k, v in value.items()}

def core(s):
    return {k: s[k] for k in ['profile', 'name', 'difficulty', 'prefs', 'wallet', 'gear', 'audio', 'profiles']}

def non_controls(s):
    return {k: v for k, v in s['cfg'].items() if k != 'controls'}

def native_events(q, spec):
    events = q['events']
    expected = []
    for kind, value, hold in spec:
        if kind == 'key':
            expected.extend([('key', value, True, hold), ('key', value, False, None)])
        else:
            expected.extend([('motion', value, None, 1), ('mouse', value, True, 2), ('mouse', value, False, None)])
    assert len(events) == len(expected)
    for i, (e, (kind, value, pressed, hold)) in enumerate(zip(events, expected)):
        assert e['sequence'] == i + 1 and e['kind'] == kind
        assert e['key' if kind == 'key' else 'target'] == value
        assert isinstance(e['frame'], int) and e['frame'] >= 0 and e['real_usec'] >= 0
        if pressed is not None:
            assert e['pressed'] is pressed
        if kind == 'key':
            assert isinstance(e['keycode'], int) and e['keycode'] > 0
            if value in ['A', 'W', 'O']:
                assert e['keycode'] == ord(value)
            if pressed:
                assert events[i + 1]['keycode'] == e['keycode']
        else:
            assert len(e['point']) == 2 and 0 <= e['point'][0] < 1280 and 0 <= e['point'][1] < 720
            if hold is not None:
                assert events[i + 1]['point'] == e['point']
        if hold is not None:
            assert events[i + 1]['frame'] - e['frame'] == hold
    assert all(b['frame'] >= a['frame'] and b['real_usec'] >= a['real_usec'] for a, b in zip(events, events[1:]))

def diagnostic(line):
    return re.sub(r'Skeleton3D:<Skeleton3D#\d+>', 'Skeleton3D:<Skeleton3D#ID>', line)
